cmake_minimum_required(VERSION 3.31.0 FATAL_ERROR)

project(C64_WIN)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)

set(CMAKE_SUPPRESS_DEVELOPER_WARNINGS TRUE)

# add dependencies
include(cmake/CPM.cmake)
include(cmake/warnings.cmake)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")

CPMAddPackage(
  NAME SDL
  GITHUB_REPOSITORY libsdl-org/SDL
  GIT_TAG release-3.2.8
)

CPMAddPackage(
  NAME imgui
  GITHUB_REPOSITORY ocornut/imgui
  GIT_TAG v1.91.9-docking
)

CPMAddPackage(
  NAME glew
  GITHUB_REPOSITORY nigels-com/glew
  GIT_TAG glew-2.2.0
)

CPMAddPackage(
  NAME spdlog
  GITHUB_REPOSITORY gabime/spdlog
  VERSION 1.15.1
  OPTIONS "SPDLOG_BUILD_SHARED OFF" "SPDLOG_BUILD_STATIC ON"
)

CPMAddPackage(
  NAME googletest
  GITHUB_REPOSITORY google/googletest
  VERSION 1.14.0
  OPTIONS "INSTALL_GTEST OFF"
)


include(FetchContent)

add_library(imgui STATIC
  ${imgui_SOURCE_DIR}/imgui.cpp
  ${imgui_SOURCE_DIR}/imgui_demo.cpp # optionally comment this out
  ${imgui_SOURCE_DIR}/imgui_draw.cpp
  ${imgui_SOURCE_DIR}/imgui_widgets.cpp
  ${imgui_SOURCE_DIR}/imgui_tables.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl3.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp
)

include_directories(imgui INTERFACE
  ${imgui_SOURCE_DIR}
  ${imgui_SOURCE_DIR}/backends
  ${SDL_SOURCE_DIR}/include
)
target_compile_definitions(imgui PUBLIC -DIMGUI_DISABLE_OBSOLETE_FUNCTIONS) # optional imgui setting
set_target_properties(imgui PROPERTIES FOLDER third-party) # optoinal IDE dir

add_subdirectory(src/C64)
# add_subdirectory(test)

file(GLOB_RECURSE UTIL_SOURCES
    "${PROJECT_SOURCE_DIR}/src/Utility/*.cpp"
    "${PROJECT_SOURCE_DIR}/src/Utility/*.hpp"
)

# add executable
add_executable(${PROJECT_NAME} 
  src/main.cpp

  src/GUI.cpp

  ${UTIL_SOURCES}
)


include(CTest)
enable_testing()

include(GoogleTest)

# link dependencies
target_link_libraries(
  ${PROJECT_NAME} 
  C64
  imgui
  opengl32
  SDL3::SDL3
  spdlog::spdlog
  stdc++
  gtest gtest_main gmock
)

target_compile_definitions(${PROJECT_NAME} PRIVATE SPDLOG_COMPILED_LIB)
file(COPY "${CMAKE_SOURCE_DIR}/roms/" DESTINATION "${EXECUTABLE_OUTPUT_PATH}/roms")


if(SDL_ADDED)
  if(WIN32 AND SDL_SHARED)
  add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${SDL_BINARY_DIR}/SDL3.dll"
      "$<TARGET_FILE_DIR:${PROJECT_NAME}>"
    COMMENT "Copying SDL3.dll to output directory"
  )
  endif()
endif()
